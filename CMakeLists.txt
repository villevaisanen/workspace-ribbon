cmake_minimum_required(VERSION 3.27)
project(workspace-ribbon VERSION 0.2.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(RIBBON_BUILD_PLUGIN "Build native Hyprland plugin" ON)
include(CTest)
if(RIBBON_BUILD_PLUGIN)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(DEPS REQUIRED IMPORTED_TARGET hyprland=0.56.2 libdrm libinput libudev pangocairo pixman-1 wayland-server xkbcommon)
  add_library(workspace-ribbon SHARED src/plugin.cpp)
  target_include_directories(workspace-ribbon PRIVATE include)
  target_compile_definitions(workspace-ribbon PRIVATE WORKSPACE_RIBBON_VERSION="${PROJECT_VERSION}")
  target_link_libraries(workspace-ribbon PRIVATE PkgConfig::DEPS)
  target_compile_options(workspace-ribbon PRIVATE -Wall -Wextra -Wno-unused-parameter)
  install(TARGETS workspace-ribbon LIBRARY DESTINATION lib)
endif()
if(BUILD_TESTING)
  add_executable(animation-test tests/animation.cpp)
  target_include_directories(animation-test PRIVATE include)
  add_test(NAME animation COMMAND animation-test)
endif()
